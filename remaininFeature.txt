```prisma
model ServiceArea {
  id              String   @id @default(cuid())
  businessId      String
  business        Business @relation(fields: [businessId], references: [id], onDelete: Cascade)

  areaName        String
  pincode         String?
  deliveryFee     Decimal?
  minimumOrder    Decimal?
  deliveryTime    String?  // "45-60 mins"

  isActive        Boolean  @default(true)
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  @@index([businessId])
  @@map("service_areas")
}

// NEW: Special hours (holidays, events)
model SpecialHours {
  id              String   @id @default(cuid())

  business        Business @relation(fields: [businessId], references: [id], onDelete: Cascade)

  name            String   // "Diwali", "New Year"
  isClosed        Boolean  @default(false)
  openTime        String?  // "10:00"
  closeTime       String?  // "22:00"

  notes           String?  @db.Text

  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  @@index([businessId])
  @@index([date])
  @@map("special_hours")
}
```

```typescript
export const serviceSettingsSchema = z.object({

  selfDelivery: z.boolean(),

  minimumOrder: z.number().min(0).optional(),
  deliveryTime: z.string().optional(),
  packagingFee: z.number().min(0).optional(),
  avgCostForTwo: z.number().min(0).optional(),
  acceptsReservations: z.boolean(),
  reservationLeadTime: z.number().min(1).max(168).optional(), // 1 hour to 1 week
  maxPartySize: z.number().min(1).max(100).optional(),
  paymentMethods: z.array(z.string()).min(1),
});

export const serviceAreaSchema = z.object({
  areaName: z.string().min(2).max(100),
  pincode: z.string().regex(/^\d{6}$/).optional(),
  deliveryFee: z.number().min(0),
  minimumOrder: z.number().min(0),
  deliveryTime: z.string(),
  isActive: z.boolean(),
});

export const specialHoursSchema = z.object({
  date: z.date(),
  name: z.string().min(2).max(100),
  isClosed: z.boolean(),
  openTime: z.string().regex(/^\d{2}:\d{2}$/).optional(),
  closeTime: z.string().regex(/^\d{2}:\d{2}$/).optional(),
  notes: z.string().max(500).optional(),
}).refine((data) => {
  if (!data.isClosed && (!data.openTime || !data.closeTime)) {
    return false;
  }
  return true;
}, "Open and close times required when not closed");
```

```prisma
model BusinessDocument {

  fileSize        Int?

  notes           String?  @db.Text

  createdAt       DateTime @default(now())

  deletedAt       DateTime?

  @@index([businessId])

model BusinessLicense {
```

#### **9.6 Booking Workflow**

```
PENDING → Confirm → CONFIRMED
          ↓
        Assign Table (if not auto-assigned)
          ↓
        Send Confirmation (SMS/Email)
          ↓
CONFIRMED → Customer Arrives → ARRIVED
          ↓
        Check-in/Seat → SEATED
          ↓
        Complete → COMPLETED

OR

Any Status → Cancel → CANCELLED
CONFIRMED → No Show → NO_SHOW
```

**Status Actions:**
- PENDING: Confirm, Cancel, Edit
- CONFIRMED: Check-in, Modify, Cancel
- ARRIVED: Seat, Cancel
- SEATED: Complete, Mark No-show
- COMPLETED: View only, Archive
- CANCELLED: View only, Re-book
- NO_SHOW: View only, Blacklist (optional)
